## DMA 中RxBufferSize 的确定

1.  BufferSize 的设计在不同场景下有所区别

   | 场景                                      | 建议 `BufferSize` 设置                      | 原因                                    |
   | ----------------------------------------- | ------------------------------------------- | --------------------------------------- |
   | **已知帧长**（比如固定 20 字节协议）      | 设置为协议帧长度                            | 方便一次 DMA 传输完成就能处理一整帧     |
   | **未知帧长，用空闲中断分帧**（UART IDLE） | 设置为比最大可能帧大一些的值，比如 256、512 | 防止溢出，同时空闲中断能提前截断        |
   | **连续流数据**（比如 ADC 采样）           | 设置为缓存一段时间的数据长度                | 结合循环模式 + 双缓冲，CPU 可以分时处理 |
   | **大块数据搬运**（比如存储器到存储器）    | 设置为总数据长度                            | 一次性搬完，减少中断次数                |

   ```c
   // 未知帧长
   #define Rx_BufferSize  256
   uint8_t Rx_Buffer[Rx_BufferSize];
   // 在DMA初始化中使用
   void Usart1_Config(){
    	// 初始化Usart 
       // 开启闲时中断
       USART_ITConfig(USARTx, USART_IT_IDLE, ENABLE);
       // NVIC 配置NVIC_InitStructure.NVIC_IRQChannel = USART1_IRQn;
   }
   
   void DMA_Config(){
           DMA_InitTypeDef dma;
   		DMA_DeInit(DMA1_Channel5);
           dma.DMA_PeripheralBaseAddr = (uint32_t)&Usart1->DR;
           dma.DMA_MemoryBaseAddr     = (uint32_t)Rx_Buffer;         // 设置缓冲区
           dma.DMA_DIR                = DMA_DIR_PeripheralSRC;       // 内存到外设传输
           dma.DMA_BufferSize         = Rx_BufferSize;               // 设置缓冲区大小
           dma.DMA_PeripheralInc      = DMA_PeripheralInc_Disable;   // 外设地址不递增
           dma.DMA_MemoryInc          = DMA_MemoryInc_Enable;        // 内存地址递增
           dma.DMA_PeripheralDataSize = DMA_PeripheralDataSize_Byte; // 外设数据大小为字节
           dma.DMA_MemoryDataSize     = DMA_MemoryDataSize_Byte;     // 内存数据大小为字节
           dma.DMA_Mode               = DMA_Mode_Normal;             // 普通模式
           dma.DMA_Priority           = DMA_Priority_High;           // 高优先级
           dma.DMA_M2M                = DMA_M2M_Disable;             // 禁用内存到内存传输
           DMA_Init(DMA1_Channel2, &dma);
   }
   
   // 在对应的UASRT1_IRQHandler 
   void USART1_IRQHandler(void)
   {
       if (USART_GetITStatus(USART1, USART_IT_IDLE) != RESET) {
           volatile uint32_t tmp;
           tmp = USART1->SR;   // 读SR清除IDLE标志第一步
           tmp = USART1->DR;   // 读DR清除IDLE标志第二步
           (void)tmp; // 防止编译报错
   
           uint16_t length = RxBuffer_Size - DMA_GetCurrDataCounter(DMA1_Channel5);
   
           DMA_Cmd(DMA1_Channel5, DISABLE);
           
           // 这里可以调用数据处理函数：process_data(rx_buf, length);
   
           DMA_SetCurrDataCounter(DMA1_Channel5, RxBuffer_Size);  // 复位计数器为缓冲区大小
           
           DMA_Cmd(DMA1_Channel5, ENABLE);
       }
   }
   ```

   ```c
   // ADC 数据采集
   void ADC1_DMA_Init(void)
   {
       // 1. 使能相关时钟
       RCC_APB2PeriphClockCmd(RCC_APB2Periph_ADC1 | RCC_APB2Periph_GPIOA, ENABLE);
       RCC_AHBPeriphClockCmd(RCC_AHBPeriph_DMA1, ENABLE);
   
       // 2. 配置 GPIOA.0 为模拟输入（ADC Channel0）
       GPIO_InitTypeDef GPIO_InitStructure;
       GPIO_InitStructure.GPIO_Pin = GPIO_Pin_0;
       GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AIN;
       GPIO_Init(GPIOA, &GPIO_InitStructure);
   
       // 3. 配置 DMA1 Channel1
       DMA_DeInit(DMA1_Channel1);
       DMA_InitTypeDef DMA_InitStructure;
       DMA_InitStructure.DMA_PeripheralBaseAddr = (uint32_t)&ADC1->DR;   // ADC 数据寄存器地址
       DMA_InitStructure.DMA_MemoryBaseAddr = (uint32_t)adc_buffer;      // 存放数据缓冲区地址
       DMA_InitStructure.DMA_DIR = DMA_DIR_PeripheralSRC;                // 外设到内存
       DMA_InitStructure.DMA_BufferSize = ADC_BUF_SIZE;                  // 缓冲区大小
       DMA_InitStructure.DMA_PeripheralInc = DMA_PeripheralInc_Disable;  // 外设地址不变
       DMA_InitStructure.DMA_MemoryInc = DMA_MemoryInc_Enable;           // 内存地址递增
       DMA_InitStructure.DMA_PeripheralDataSize = DMA_PeripheralDataSize_HalfWord; // 16位
       DMA_InitStructure.DMA_MemoryDataSize = DMA_MemoryDataSize_HalfWord;         // 16位
       DMA_InitStructure.DMA_Mode = DMA_Mode_Circular;                    // 循环模式
       DMA_InitStructure.DMA_Priority = DMA_Priority_High;               // 高优先级
       DMA_InitStructure.DMA_M2M = DMA_M2M_Disable;                       // 非内存到内存
   
       DMA_Init(DMA1_Channel1, &DMA_InitStructure);
       DMA_Cmd(DMA1_Channel1, ENABLE);
   
       // 4. 配置 ADC1
       ADC_InitTypeDef ADC_InitStructure;
       ADC_InitStructure.ADC_Mode = ADC_Mode_Independent;
       ADC_InitStructure.ADC_ScanConvMode = DISABLE;            // 单通道
       ADC_InitStructure.ADC_ContinuousConvMode = ENABLE;       // 连续转换
       ADC_InitStructure.ADC_ExternalTrigConv = ADC_ExternalTrigConv_None;  // 软件触发
       ADC_InitStructure.ADC_DataAlign = ADC_DataAlign_Right;   // 右对齐
       ADC_InitStructure.ADC_NbrOfChannel = 1;                   // 通道数1
   
       ADC_Init(ADC1, &ADC_InitStructure);
   
       ADC_RegularChannelConfig(ADC1, ADC_Channel_0, 1, ADC_SampleTime_55Cycles5);
   
       ADC_DMACmd(ADC1, ENABLE);    // 使能 ADC DMA 请求
       ADC_Cmd(ADC1, ENABLE);       // 使能 ADC
   
       // 5. ADC 校准
       ADC_ResetCalibration(ADC1);
       while(ADC_GetResetCalibrationStatus(ADC1));
       ADC_StartCalibration(ADC1);
       while(ADC_GetCalibrationStatus(ADC1));
   
       // 6. 软件启动 ADC 转换
       ADC_SoftwareStartConvCmd(ADC1, ENABLE);
   }
   ```
